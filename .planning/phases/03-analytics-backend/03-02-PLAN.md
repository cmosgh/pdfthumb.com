    **Step 1 — Create `usage-tracking.interceptor.ts`:**

    ```typescript
    import {
      Injectable,
      NestInterceptor,
      ExecutionContext,
      CallHandler,
      Logger,
    } from '@nestjs/common';
    import { InjectQueue } from '@nestjs/bullmq';
    import { Queue } from 'bullmq';
    import { Observable } from 'rxjs';
    import { tap, catchError } from 'rxjs/operators';
    import { throwError } from 'rxjs';

    function deriveErrorCode(error: unknown): string {
      const msg = (error as Error)?.message ?? '';
      if (msg.includes('PDF')) return 'INVALID_PDF';
      if (msg.includes('timeout') || (error as NodeJS.ErrnoException)?.code === 'ETIMEDOUT') return 'TIMEOUT';
      return 'UNKNOWN';
    }

    @Injectable()
    export class UsageTrackingInterceptor implements NestInterceptor {
      private readonly logger = new Logger(UsageTrackingInterceptor.name);

      constructor(@InjectQueue('usage-events') private readonly queue: Queue) {}

      intercept(context: ExecutionContext, next: CallHandler): Observable<unknown> {
        const request = context.switchToHttp().getRequest<{
          user?: { id?: string };
          headers: Record<string, string | undefined>;
        }>();
        const startTime = Date.now();
        const userId = request.user?.id ?? null;
        const apiKeyRaw = request.headers['x-api-key'] ?? null;

        // Derive prefix/suffix for async processor lookup (avoid storing raw key)
        const prefix = apiKeyRaw ? apiKeyRaw.substring(0, 4) : null;
        const suffix = apiKeyRaw ? apiKeyRaw.substring(apiKeyRaw.length - 4) : null;

        return next.handle().pipe(
          tap(() => {
            const durationMs = Date.now() - startTime;
            const response = context.switchToHttp().getResponse<{ statusCode: number }>();
            const statusCode = response.statusCode;

            this.queue
              .add('track-usage', {
                userId,
                prefix,
                suffix,
                statusCode,
                success: statusCode < 400,
                errorCode: null,
                durationMs,
                timestamp: new Date().toISOString(),
              })
              .catch((err: unknown) => {
                // Swallow — tracking failure MUST NOT affect thumbnail response
                this.logger.warn('Failed to enqueue usage event', err);
              });
          }),
          catchError((error: unknown) => {
            const durationMs = Date.now() - startTime;
            this.queue
              .add('track-usage', {
                userId,
                prefix,
                suffix,
                statusCode: (error as { status?: number })?.status ?? 500,
                success: false,
                errorCode: deriveErrorCode(error),
                durationMs,
                timestamp: new Date().toISOString(),
              })
              .catch((err: unknown) => {
                this.logger.warn('Failed to enqueue error usage event', err);
              });
            return throwError(() => error); // Re-throw so original error response is sent
          }),
        );
      }
    }
    ```

    CRITICAL: Use `tap()` not `map()` — `tap()` is a side effect that passes the value through unchanged. Using `map()` would corrupt the thumbnail response body.
    CRITICAL: The `catchError` callback MUST call `throwError(() => error)` after enqueuing — if it does not re-throw, the client receives 200 for all errors.

    **Step 2 — Create `analytics.processor.ts`:**

    ```typescript
    import { Processor, WorkerHost, OnWorkerEvent } from '@nestjs/bullmq';
    import { InjectQueue } from '@nestjs/bullmq';
    import { Job, Queue } from 'bullmq';
    import { InjectRepository } from '@nestjs/typeorm';
    import { Repository } from 'typeorm';
    import { Logger } from '@nestjs/common';
    import { UsageEvent } from './entities/usage-event.entity';
    import { ApiKeyService } from '../api-key/api-key.service';

    @Processor('usage-events', { concurrency: 5 })
    export class AnalyticsProcessor extends WorkerHost {
      private readonly logger = new Logger(AnalyticsProcessor.name);

      constructor(
        @InjectRepository(UsageEvent)
        private readonly usageEventRepo: Repository<UsageEvent>,
        @InjectQueue('usage-events-dlq')
        private readonly dlqQueue: Queue,
        private readonly apiKeyService: ApiKeyService,
      ) {
        super();
      }

      async process(job: Job): Promise<void> {
        const { userId, prefix, suffix, statusCode, success, errorCode, durationMs, timestamp } =
          job.data as {
            userId: string | null;
            prefix: string | null;
            suffix: string | null;
            statusCode: number;
            success: boolean;
            errorCode: string | null;
            durationMs: number;
            timestamp: string;
          };

        // Resolve api_key_id asynchronously — inside the job so no HTTP latency hit
        let apiKeyId: string | null = null;
        if (prefix && suffix) {
          const keyEntity = await this.apiKeyService.findApiKeyEntityByPrefixSuffix(prefix, suffix);
          apiKeyId = keyEntity?.id ?? null;
        }

        const event = this.usageEventRepo.create({
          userId,
          apiKeyId,
          statusCode,
          success,
          errorCode: errorCode ?? null,
          durationMs,
          createdAt: new Date(timestamp),
        });
        await this.usageEventRepo.save(event);
      }

      @OnWorkerEvent('failed')
      async onFailed(job: Job, err: Error): Promise<void> {
        const maxAttempts = job.opts.attempts ?? 5;
        if (job.attemptsMade >= maxAttempts) {
          this.logger.error(
            `Job ${job.id} permanently failed after ${job.attemptsMade} attempts: ${err.message}`,
          );
          await this.dlqQueue.add('failed-event', job.data).catch(() => {
            this.logger.error('Failed to move job to DLQ — data may be lost');
          });
        }
      }
    }
    ```

    **Step 3 — Create `analytics.module.ts`:**

    ```typescript
    import { Module } from '@nestjs/common';
    import { BullModule } from '@nestjs/bullmq';
    import { TypeOrmModule } from '@nestjs/typeorm';
    import { UsageEvent } from './entities/usage-event.entity';
    import { AnalyticsProcessor } from './analytics.processor';
    import { UsageTrackingInterceptor } from './usage-tracking.interceptor';
    import { ApiKeyModule } from '../api-key/api-key.module';

    const defaultJobOptions = {
      attempts: 5,
      backoff: {
        type: 'exponential' as const,
        delay: 1000, // base: 1s → 2s → 4s → 8s → 16s
      },
      removeOnComplete: { count: 1000 },
      removeOnFail: false, // Keep failed jobs for DLQ inspection
    };

    @Module({
      imports: [
        BullModule.registerQueue(
          { name: 'usage-events', defaultJobOptions },
          { name: 'usage-events-dlq' },
        ),
        TypeOrmModule.forFeature([UsageEvent]),
        ApiKeyModule,
      ],
      providers: [
        AnalyticsProcessor,
        UsageTrackingInterceptor,
      ],
      exports: [UsageTrackingInterceptor],
    })
    export class AnalyticsModule {}
    ```

    Note: `AnalyticsModule` exports `UsageTrackingInterceptor` so `ThumbnailController` (in `AppModule` scope) can use it. The controller is declared in `AppModule`, which imports `AnalyticsModule` — DI will resolve.
