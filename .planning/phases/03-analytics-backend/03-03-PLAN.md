    **Step 2 — Create `src/analytics/analytics.service.ts`:**

    ```typescript
    import { Injectable } from '@nestjs/common';
    import { InjectRepository } from '@nestjs/typeorm';
    import { Repository } from 'typeorm';
    import { UsageEvent } from './entities/usage-event.entity';
    import {
      UserSummaryResponse,
      AdminSummaryResponse,
      DailyBucket,
    } from './dto/analytics-summary.dto';

    @Injectable()
    export class AnalyticsService {
      private readonly isProduction = process.env.NODE_ENV === 'production';

      constructor(
        @InjectRepository(UsageEvent)
        private readonly usageEventRepo: Repository<UsageEvent>,
      ) {}

      async getUserDailySummary(
        userId: string,
        days: number,
      ): Promise<UserSummaryResponse> {
        const cutoff = this.getCutoffDate(days);
        const dateTruncExpr = this.dateTruncExpr();

        const rows = await this.usageEventRepo.query(
          `SELECT
             ${dateTruncExpr} AS date,
             COUNT(*) AS call_count,
             SUM(CASE WHEN success = ${this.isProduction ? 'false' : '0'} THEN 1 ELSE 0 END) AS error_count,
             AVG("durationMs") AS avg_duration_ms,
             COUNT(DISTINCT "apiKeyId") AS unique_api_keys
           FROM usage_events
           WHERE "userId" = $1
             AND "createdAt" >= $2
           GROUP BY ${dateTruncExpr}
           ORDER BY ${dateTruncExpr} ASC`,
          [userId, cutoff.toISOString()],
        ) as DailyBucket[];

        return { dailyBuckets: rows };
      }

      async getAdminDailySummary(days: number): Promise<AdminSummaryResponse> {
        const cutoff = this.getCutoffDate(days);
        const dateTruncExpr = this.dateTruncExpr();

        const dailyBuckets = (await this.usageEventRepo.query(
          `SELECT
             ${dateTruncExpr} AS date,
             COUNT(*) AS call_count,
             SUM(CASE WHEN success = ${this.isProduction ? 'false' : '0'} THEN 1 ELSE 0 END) AS error_count,
             AVG("durationMs") AS avg_duration_ms,
             COUNT(DISTINCT "apiKeyId") AS unique_api_keys
           FROM usage_events
           WHERE "createdAt" >= $1
           GROUP BY ${dateTruncExpr}
           ORDER BY ${dateTruncExpr} ASC`,
          [cutoff.toISOString()],
        )) as DailyBucket[];

        const topUsersByVolume = (await this.usageEventRepo.query(
          `SELECT
             e."userId",
             u."email",
             COUNT(*) AS total_calls
           FROM usage_events e
           LEFT JOIN users u ON u.id = e."userId"
           WHERE e."createdAt" >= $1
             AND e."userId" IS NOT NULL
           GROUP BY e."userId", u."email"
           ORDER BY total_calls DESC
           LIMIT 10`,
          [cutoff.toISOString()],
        )) as Array<{ userId: string; email: string; totalCalls: number }>;

        const topUsersByErrors = (await this.usageEventRepo.query(
          `SELECT
             e."userId",
             u."email",
             COUNT(*) AS error_count
           FROM usage_events e
           LEFT JOIN users u ON u.id = e."userId"
           WHERE e."createdAt" >= $1
             AND e."userId" IS NOT NULL
             AND e.success = ${this.isProduction ? 'false' : '0'}
           GROUP BY e."userId", u."email"
           ORDER BY error_count DESC
           LIMIT 10`,
          [cutoff.toISOString()],
        )) as Array<{ userId: string; email: string; errorCount: number }>;

        return { dailyBuckets, topUsersByVolume, topUsersByErrors };
      }

      private getCutoffDate(days: number): Date {
        const cutoff = new Date();
        cutoff.setDate(cutoff.getDate() - days);
        return cutoff;
      }

      private dateTruncExpr(): string {
        return this.isProduction
          ? `DATE_TRUNC('day', "createdAt")`
          : `strftime('%Y-%m-%d', "createdAt")`;
      }
    }
    ```

    IMPORTANT about SQLite `success` boolean: SQLite stores booleans as 0/1 integers. The `success = false` comparison used in Postgres raw SQL fails silently in SQLite — it must be `success = 0`. The `isProduction` branch above handles this: Postgres uses `false`, SQLite uses `0`.

    IMPORTANT about `DATE_TRUNC` vs `strftime`: These return different types. Postgres `DATE_TRUNC` returns a timestamp. SQLite `strftime` returns a string. Both are acceptable for bucketing — the response `date` field will be a timestamp string in prod and a `YYYY-MM-DD` string in dev. Frontend will handle both.

    **Step 1 — Create `src/analytics/analytics.controller.ts`:**

    Follow the EXACT guard pattern from Phase 1:
    - User endpoint: `JwtOrApiKeyAuthGuard` (same as `ThumbnailController` — API keys can call analytics)
    - Admin endpoint: `JwtAuthGuard` + `RolesGuard` + `@Roles(UserRole.ADMIN)` (same as `AdminController.ping`)

    ```typescript
    import {
      Controller,
      Get,
      Query,
      UseGuards,
      Request,
    } from '@nestjs/common';
    import { DefaultValuePipe, ParseIntPipe } from '@nestjs/common';
    import { JwtOrApiKeyAuthGuard } from '../common/guards/jwt-or-api-key.guard';
    import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
    import { RolesGuard } from '../auth/guards/roles.guard';
    import { Roles } from '../auth/decorators/roles.decorator';
    import { UserRole } from '../auth/entities/user.entity';
    import { AnalyticsService } from './analytics.service';

    @Controller('analytics')
    export class AnalyticsController {
      constructor(private readonly analyticsService: AnalyticsService) {}

      @Get('summary')
      @UseGuards(JwtOrApiKeyAuthGuard)
      async getUserSummary(
        @Request() req: { user: { id: string } },
        @Query('days', new DefaultValuePipe(30), ParseIntPipe) days: number,
      ) {
        return this.analyticsService.getUserDailySummary(req.user.id, days);
      }

      @Get('admin/summary')
      @UseGuards(JwtAuthGuard, RolesGuard)
      @Roles(UserRole.ADMIN)
      async getAdminSummary(
        @Query('days', new DefaultValuePipe(30), ParseIntPipe) days: number,
      ) {
        return this.analyticsService.getAdminDailySummary(days);
      }
    }
    ```

    CRITICAL: `DefaultValuePipe` MUST precede `ParseIntPipe` in the pipe array. If they are reversed, a missing `?days` param causes `ParseIntPipe` to receive `undefined` and throw `BadRequestException`.

    CRITICAL: Admin endpoint uses `JwtAuthGuard` (Bearer only) — NOT `JwtOrApiKeyAuthGuard`. Admin actions should never be callable via API keys. This matches `AdminController.ping` pattern from Phase 1.

    ```typescript
    import { BullModule } from '@nestjs/bullmq';
    import { TypeOrmModule } from '@nestjs/typeorm';
    import { UsageEvent } from './entities/usage-event.entity';
    import { AnalyticsProcessor } from './analytics.processor';
    import { UsageTrackingInterceptor } from './usage-tracking.interceptor';
    import { AnalyticsService } from './analytics.service';
    import { AnalyticsController } from './analytics.controller';
    import { ApiKeyModule } from '../api-key/api-key.module';
    import { AuthModule } from '../auth/auth.module';
    import { CommonModule } from '../common/common.module';
    import { RolesGuard } from '../auth/guards/roles.guard';

    const defaultJobOptions = {
      attempts: 5,
      backoff: {
        type: 'exponential' as const,
        delay: 1000,
      },
      removeOnComplete: { count: 1000 },
      removeOnFail: false,
    };

    @Module({
      imports: [
        BullModule.registerQueue(
          { name: 'usage-events', defaultJobOptions },
          { name: 'usage-events-dlq' },
        ),
        TypeOrmModule.forFeature([UsageEvent]),
        ApiKeyModule,
        AuthModule,
        CommonModule,
      ],
      controllers: [AnalyticsController],
      providers: [
        AnalyticsProcessor,
        UsageTrackingInterceptor,
        AnalyticsService,
        RolesGuard,
      ],
      exports: [UsageTrackingInterceptor],
    })
    export class AnalyticsModule {}
    ```

    Note: This replaces the skeleton `analytics.module.ts` created in plan 03-02. The `defaultJobOptions` constant is duplicated here intentionally to keep the module self-contained — no shared config import needed.
