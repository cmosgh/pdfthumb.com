2. Update all `setAuthState(...)` calls that set the full object to include `isRoleLoading: false` (login, logout, error paths in refreshToken, loadAuthState paths). The `login()` function should set `isRoleLoading: true` (role fetch is about to start) — do NOT set roles in login(), roles come from fetchMe().

3. Add `fetchMe` function using `useCallback`:
```typescript
const fetchMe = useCallback(async (accessToken: string): Promise<void> => {
  setAuthState(prev => ({ ...prev, isRoleLoading: true }));
  let lastError: Error | null = null;
  for (let attempt = 0; attempt < 2; attempt++) {
    try {
      const profile = await authApi.me(accessToken);
      setAuthState(prev => ({
        ...prev,
        isRoleLoading: false,
        user: prev.user
          ? { ...prev.user, roles: profile.roles }
          : null,
      }));
      return;
    } catch (err) {
      lastError = err as Error;
      if (attempt < 1) await new Promise(r => setTimeout(r, 1000));
    }
  }
  // All retries failed — log out per user decision
  console.error('fetchMe failed after retries:', lastError);
  setAuthState(prev => ({ ...prev, isRoleLoading: false }));
  await logout();
}, [logout]);
```
