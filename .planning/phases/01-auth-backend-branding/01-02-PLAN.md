import { Controller, Get, UseGuards } from '@nestjs/common';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { RolesGuard } from '../auth/guards/roles.guard';
import { Roles } from '../auth/decorators/roles.decorator';
import { UserRole } from '../auth/entities/user.entity';

@Controller('admin')
export class AdminController {
  @Get('ping')
  @UseGuards(JwtAuthGuard, RolesGuard)
  @Roles(UserRole.ADMIN)
  ping(): { status: string } {
    return { status: 'ok' };
  }
}
```

Guard order is mandatory: `@UseGuards(JwtAuthGuard, RolesGuard)` — JwtAuthGuard FIRST. If reversed, RolesGuard runs before req.user is set and all requests get 401 instead of 403.

import { RolesGuard } from './roles.guard';
import { Reflector } from '@nestjs/core';
import { ExecutionContext, ForbiddenException } from '@nestjs/common';
import { UserRole } from '../entities/user.entity';

describe('RolesGuard', () => {
  let guard: RolesGuard;
  let reflector: Reflector;

  beforeEach(() => {
    reflector = new Reflector();
    guard = new RolesGuard(reflector);
  });

  const mockContext = (
    userRoles: UserRole[],
    requiredRoles: UserRole[],
  ): ExecutionContext => {
    jest.spyOn(reflector, 'getAllAndOverride').mockReturnValue(requiredRoles);
    return {
      switchToHttp: () => ({
        getRequest: () => ({ user: { roles: userRoles } }),
      }),
      getHandler: () => jest.fn(),
      getClass: () => jest.fn(),
    } as unknown as ExecutionContext;
  };

  it('allows access when user has required role', () => {
    const ctx = mockContext([UserRole.ADMIN], [UserRole.ADMIN]);
    expect(guard.canActivate(ctx)).toBe(true);
  });

  it('throws ForbiddenException when user lacks required role', () => {
    const ctx = mockContext([UserRole.USER], [UserRole.ADMIN]);
    expect(() => guard.canActivate(ctx)).toThrow(ForbiddenException);
  });

  it('allows access when no roles required (route has no @Roles decorator)', () => {
    const ctx = mockContext([UserRole.USER], []);
    expect(guard.canActivate(ctx)).toBe(true);
  });

  it('allows access when user has one of multiple required roles', () => {
    const ctx = mockContext([UserRole.ADMIN], [UserRole.USER, UserRole.ADMIN]);
    expect(guard.canActivate(ctx)).toBe(true);
  });
});
```
